5/// Largest heap handed to talc in one claim: `isize::MAX` rounded down to a page.
6///
7/// talc walks a free gap with `base.add(size)` / `acme.sub(size)`, which is undefined
8/// behaviour once the gap exceeds `isize::MAX`: 2 GiB on riscv32, which an arena on
9/// the 4 GiB machine image (`RamSize::FourGb`) can exceed. The arena is therefore
10/// claimed as several adjacent heaps of equal size. talc never merges across heaps
11/// (each heap's bottom is tagged allocated), so every gap stays within its heap, and
12/// one allocation is limited to a single heap: about half the arena once the arena
13/// exceeds `isize::MAX`, which can be well below the `Layout` cap.
14const MAX_HEAP_SIZE: usize = (isize::MAX as usize) & PAGE_MASK;
15
16/// Granularity of heap sizes. talc only needs its own word-sized chunk alignment;
17/// a page keeps every heap boundary as aligned as the arena start.
18const PAGE: usize = 4096;
19
20/// Clears the in-page offset bits: `x & PAGE_MASK` rounds `x` down to a page.
21const PAGE_MASK: usize = !(PAGE - 1);
22
23pub struct TalcAllocator {
24    state: UnsafeCell<TalcState>,
25}
26
27struct TalcState {
28    allocator: Option<talc::Talc<talc::ClaimOnOom>>,
29}
30
31unsafe impl Sync for TalcAllocator {}
32
33impl TalcAllocator {
34    pub const fn uninit() -> Self {
35        Self {
36            state: UnsafeCell::new(TalcState { allocator: None }),
37        }
38    }
39
40    /// # Safety
41    ///
42    /// Caller must ensure `start` and `end` define a writable heap range.
43    pub unsafe fn init(&self, start: *mut usize, end: *mut usize) {
44        self.init_with_max_heap_size(start, end, MAX_HEAP_SIZE);
45    }
46
47    /// # Safety
48    ///
49    /// Same as [`Self::init`]. Additionally, `max` must be a multiple of `PAGE` and at
50    /// most `MAX_HEAP_SIZE`: heap sizes are rounded up to a page, and a heap larger
51    /// than `isize::MAX` is undefined behaviour inside talc.
52    unsafe fn init_with_max_heap_size(&self, start: *mut usize, end: *mut usize, max: usize) {
53        debug_assert!(max.is_multiple_of(PAGE) && max <= MAX_HEAP_SIZE);
54        let state = &mut *self.state.get();
55        let end = end as usize;
56        let mut base = start as usize;
57        if end <= base {
58            state.allocator = None;
59            return;
60        }
61
62        // Equal heaps rather than `max`-sized ones plus a remainder: a remainder
63        // can be too small for talc to claim. Rounding up to a page keeps each
64        // heap within `max`, and the last one within a few pages of the others.
65        let total = end - base;
66        let heap_size = total.div_ceil(total.div_ceil(max)).next_multiple_of(PAGE);
67
68        let mut allocator = talc::Talc::new(talc::ClaimOnOom::new(talc::Span::empty()));
69        while base < end {
70            let size = (end - base).min(heap_size);
71            let span = talc::Span::from_base_size(base as *mut u8, size);
72            allocator.claim(span).expect("must claim heap span");
73            base += size;
74        }
75        state.allocator = Some(allocator);
76    }
77
78    unsafe fn alloc_inner(&self, layout: Layout) -> *mut u8 {
79        let state = &mut *self.state.get();
80        let Some(allocator) = state.allocator.as_mut() else {
81            return null_mut();
82        };
83        allocator
84            .malloc(layout)
85            .map_or(null_mut(), |nn| nn.as_ptr())
86    }
87
88    unsafe fn dealloc_inner(&self, ptr: *mut u8, layout: Layout) {
89        if ptr.is_null() || layout.size() == 0 {
90            return;
91        }
92
93        let state = &mut *self.state.get();
94        if let Some(allocator) = state.allocator.as_mut() {
95            allocator.free(NonNull::new_unchecked(ptr), layout);
96        }
97    }
98}
99
100unsafe impl GlobalAlloc for TalcAllocator {
101    unsafe fn alloc(&self, layout: Layout) -> *mut u8 {
102        self.alloc_inner(layout)
103    }
104
105    unsafe fn dealloc(&self, ptr: *mut u8, layout: Layout) {
106        self.dealloc_inner(ptr, layout)
107    }
108
109    // Deliberately alloc + copy + free rather than talc's `grow_in_place`, which
110    // compares addresses and, near the top of a 32-bit address space, grants a growth
111    // whose end wraps past 2^32 (SFBdragon/talc#56).
112    unsafe fn realloc(&self, ptr: *mut u8, old_layout: Layout, new_size: usize) -> *mut u8 {
113        // Safety for `from_size_align_unchecked`: `old_layout` _must_ be the same layout that was
114        // used to allocate `ptr`.`
115
116        if ptr.is_null() {
117            return self.alloc_inner(Layout::from_size_align_unchecked(
118                new_size,
119                old_layout.align(),
120            ));
121        }
122
123        let new_layout = Layout::from_size_align_unchecked(new_size, old_layout.align());
124        let new_ptr = self.alloc_inner(new_layout);
125        if !new_ptr.is_null() {
126            let copy_len = core::cmp::min(old_layout.size(), new_size);
127            core::ptr::copy_nonoverlapping(ptr, new_ptr, copy_len);
128            self.dealloc_inner(ptr, old_layout);
129        }
130        new_ptr
131    }
132}

152#[cfg(test)]

154    use super::*;

157    const ARENA: usize = 1 << 20;
158    const HEAP: usize = 256 << 10;
159
160    /// A 1 MiB host buffer claimed as 256 KiB heaps, so the split is exercised on a
161    /// 64-bit host where `MAX_HEAP_SIZE` is out of reach.
162    fn split_allocator(arena: &mut [u64]) -> (TalcAllocator, usize, usize) {
163        let allocator = TalcAllocator::uninit();
164        let start = arena.as_mut_ptr() as usize;
165        let end = start + core::mem::size_of_val(arena);
166        unsafe { allocator.init_with_max_heap_size(start as *mut usize, end as *mut usize, HEAP) };
167        (allocator, start, end)
168    }
169
170    #[test]
171    fn allocations_across_heaps_stay_in_the_arena_and_do_not_overlap() {
172        let mut arena = vec![0u64; ARENA / 8];
173        let (allocator, start, end) = split_allocator(&mut arena);
174        let layout = Layout::from_size_align(16 << 10, 8).unwrap();
175
176        let mut blocks = Vec::new();
177        loop {
178            let ptr = unsafe { allocator.alloc(layout) };
179            if ptr.is_null() {
180                break;
181            }
182            let addr = ptr as usize;
183            assert!(addr >= start && addr + layout.size() <= end);
184            blocks.push(addr);
185        }
186        // Each 256 KiB heap loses a little to tags (and the first to talc's
187        // metadata), so a few fewer than 64 fit, spread over all four heaps.
188        assert!(blocks.len() >= 56, "only {} blocks fit", blocks.len());
189        assert!(blocks.iter().any(|&a| a >= start + 3 * HEAP));
190
191        blocks.sort_unstable();
192        assert!(blocks.windows(2).all(|w| w[0] + layout.size() <= w[1]));
193
194        for &addr in &blocks {
195            unsafe { allocator.dealloc(addr as *mut u8, layout) };
196        }
197        let again = unsafe { allocator.alloc(layout) };
198        assert!(!again.is_null());
199    }
200
201    #[test]
202    fn an_arena_just_past_a_multiple_of_the_heap_limit_is_claimed_in_full() {
203        // Four heaps' worth plus one word: split greedily, the last heap would be
204        // eight bytes, which talc cannot claim.
205        let mut arena = vec![0u64; ARENA / 8 + 1];
206        let (allocator, start, end) = split_allocator(&mut arena);
207
208        let layout = Layout::from_size_align(HEAP / 2, 8).unwrap();
209        let mut blocks = Vec::new();
210        loop {
211            let ptr = unsafe { allocator.alloc(layout) };
212            if ptr.is_null() {
213                break;
214            }
215            assert!(ptr as usize >= start && ptr as usize + layout.size() <= end);
216            blocks.push(ptr);
217        }
218        // Five heaps of 192-208 KiB each hold one 128 KiB block apiece.
219        assert_eq!(blocks.len(), 5);
220    }
221
222    #[test]
223    fn an_allocation_larger_than_one_heap_is_refused() {
224        let mut arena = vec![0u64; ARENA / 8];
225        let (allocator, _, _) = split_allocator(&mut arena);
226
227        let fits = Layout::from_size_align(HEAP / 2, 8).unwrap();
228        let straddles = Layout::from_size_align(HEAP + (16 << 10), 8).unwrap();
229        assert!(!unsafe { allocator.alloc(fits) }.is_null());
230        assert!(unsafe { allocator.alloc(straddles) }.is_null());
231    }
